# Sankalp IIB: PySpark Silver Layer Transformation & Validation Check
import sys
from awsglue.transforms import *
from awsglue.utils import getResolvedOptions
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from pyspark.sql.functions import col, when, to_date, current_timestamp

glueContext = GlueContext(SparkContext.getOrCreate())
spark = glueContext.spark_session

# Ingest Bronze policy & claims dataset
raw_df = spark.read.parquet("s3://chola-sankalp-bronze/policy_data/")

# Schema enforcement & validation logic
validated_df = raw_df.filter(col("policy_no").isNotNull() & (col("premium_amt") >= 0)) \
    .withColumn("issue_date", to_date(col("issue_date_raw"), "yyyy-MM-dd")) \
    .withColumn("ingest_timestamp", current_timestamp()) \
    .dropDuplicates(["policy_no", "endorsement_id"])

# Write to Silver layer partitioned by business date
validated_df.write.mode("append") \
    .partitionBy("issue_date") \
    .parquet("s3://chola-sankalp-silver/curated_policy/")

-- Redshift Materialized View & Multi-Tenant Settlement Summary
CREATE MATERIALIZED VIEW mv_post_trade_settlement_summary AS
SELECT 
    t.client_id,
    t.trade_date,
    t.settlement_status,
    COUNT(t.trade_id) AS total_trades,
    SUM(t.trade_value_usd) AS aggregate_volume_usd,
    AVG(DATEDIFF(minute, t.trade_timestamp, t.settled_timestamp)) AS avg_settlement_latency_mins
FROM dw_gold.fact_trade_events t
WHERE t.trade_date >= DATEADD(day, -90, CURRENT_DATE)
GROUP BY t.client_id, t.trade_date, t.settlement_status;

-- Automated Refresh Stored Procedure for Operational Pipeline
CREATE OR REPLACE PROCEDURE prc_refresh_settlement_views()
LANGUAGE plpgsql
AS $$
BEGIN
    REFRESH MATERIALIZED VIEW mv_post_trade_settlement_summary;
    COMMIT;
END;
$$;
